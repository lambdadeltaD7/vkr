	\section{Обозначения}
	Фиксируем $x \in \mathbb{R}$\\
	Пусть $f(x) = (1, x, \ldots, x^m)^\mathrm{T}$, $c=f'(x)$\\
	
	\noindent Рассмотрим план
	\[
	\xi = 
	\left(
	\begin{matrix}
		x_1 \ x_2 \ldots x_n \\
		\omega_1 \ \omega_2 \ldots \omega_n	
	\end{matrix}
	\right)
	\]
	
	\noindent Та самая система для плана с $n$ узлами имеет вид:
	\[
	c = h M(\xi) p = h \sum_{k=1}^{n} \omega_k <p,f(x_k)> f(x_k)
	\]	
	
	\noindent или, c учётом $h = (\lambda ^*)^{-1}$ и чередования знаков т.е. $<p,f(x_k)> = (-1)^{k+1}$
	\[
	\lambda ^*
	\begin{pmatrix}
		0 \\
		1 \\
		\vdots \\
		m x^{m-1}
	\end{pmatrix} = 
	\sum_{k=1}^{n} a_k 
	\begin{pmatrix}
		1 \\
		x_k \\
		\vdots \\
		x_k^{m}
	\end{pmatrix}, \quad a_k = \omega_k <p,f(x_k)>
	\]
	
	\noindent Введём линейные операторы $A_1, A_2, A: \mathbb{P}_m \to \mathbb{R}$:
	\[
	A_1(p) := \lambda^* p'(x)
	\]
	\[
	A_2(p) := \sum_{k=1}^{n} a_k p(x_k)
	\]
	\[
	A(p) := A_1(p) - A_2(p)
	\]
	
	\noindent Нетрудно видеть, что наша система эквивалентна вот такому:
	\[
	\lbrace 1, x, \ldots, x^m \rbrace \subset \ker A
	\]
	
	\noindent Т.к $A$ линейно, то это в свою очередь эквивалентно
	\begin{equation}
		\mathbb{P}_m \subset \ker A
		\label{eq:cond1}
	\end{equation}

	\section{Чебышёвские планы}
	\subsection{Ищем веса в чебышёвских планах}
	
	
	\noindent (\textbf{в этом разделе} $\mathbf{n = m+1}$)\\ 
	
	
	\noindent Вспоминаем, что \eqref{eq:lagr_expansion}  $\forall p \in \mathbb{P}_m$ даёт нам
	\[
	p(t) = 
	\sum_{k=1}^{m+1} p(x_k) L_k(t), \  \forall t \in \mathbb{R}
	\]
	
	\noindent Дифференцируя это равенство по $t$ и подставляя $t=x$ получаем 
	\[
	p'(x) = 
	\sum_{k=1}^{m+1} p(x_k) L_k'(x)
	\]
	
	
	\noindent Получили, что $\forall p \in \mathbb{P}_m$
	\begin{equation}
		\lambda^* p'(x) = 
		\sum_{k=1}^{m+1} p(x_k) ( \lambda^* L_k'(x) ) 
		\label{eq:re1_2}
	\end{equation}
	
	\noindent Но из условия \eqref{eq:cond1} 
	\[
	A(p) = 0, \quad \forall p \in \mathbb{P}_m
	\]
	
	\noindent То есть $\forall p \in \mathbb{P}_m$
	\begin{equation}
		\lambda^* p'(x) = 
		\sum_{k=1}^{m+1} p(x_k) ( a_k ) 
		\label{eq:re2_2}
	\end{equation}
	
	\noindent Из \eqref{eq:re1_2} и \eqref{eq:re2_2} получаем $\forall p \in \mathbb{P}_m$
	\[
	\sum_{k=1}^{m+1} p(x_k) ( a_k ) = \sum_{k=1}^{m+1} p(x_k) ( \lambda^* L_k'(x) ) 
	\] 
	
	\noindent Выбирая в качестве $p$ штуки которые равны $1$ и $0$ где надо ($L_k(x) ???$), делаем вывод
	\[
	a_k = \lambda^* L_k'(x), \ \forall k \in 1 \ldots m+1 
	\]
	\[
	\omega_k (-1)^{k+1} = \lambda^* L_k'(x), \ \forall k \in 1 \ldots m+1 
	\]
	\[
	\omega_k  = (-1)^{k+1} \lambda^* L_k'(x), \ \forall k \in 1 \ldots m+1 
	\]
	
	
	\noindent Юзая условие на веса $\sum_{k=1}^{m+1} \omega_k = 1$ ищем лямбду:
	\[
	1 = \sum_{k=1}^{m+1} \omega_k = \sum_{k=1}^{m+1} (-1)^{k+1} \lambda^* L_k'(x) = \lambda^* \sum_{k=1}^{m+1} (-1)^{k+1}  L_k'(x)
	\]
	
	\noindent получаем
	\[
	\lambda^* = \left( \sum_{k=1}^{m+1} (-1)^{k+1}  L_k'(x) \right) ^ {-1}
	\]
	
	\noindent финальное выражение для весов
	\[
	\omega_k  = 
	\frac
	{
		(-1)^{k+1} L_k'(x)
	}
	{
		\sum_{i=1}^{m+1} (-1)^{i+1}  L_i'(x)
	}, \ \forall k \in 1 \ldots m+1 
	\]
	
	\[
	L_k(x) = \underset{k \neq j}{\prod_{j=1}^{m+1}} \frac{x - x_j}{x_k - x_j}, \ k \in 1 \ldots m+1
	\]
	
	\subsection{Ограничение на $x$ от весов}
	
	Мы требуем $\forall k \in 1 \ldots m+1 \ \omega_k > 0 $
	
	\noindent То есть нужно решить систему неравенств относительно $x$:
	\[
	\omega_k  = (-1)^{k+1} \lambda^* L_k'(x)  > 0 , \ \forall k \in 1 \ldots m+1 
	\]
	
	
	
	\subsection{Ищем узлы в чебышёвских планах}
	
	\noindent А узлы это просто точки альтернанса полинома Чебышёва 1-го рода 
	\[
	T_{m}(x) = \cos( m \arccos(x))
	\]
	\noindent Они имеют вид 
	\[
	x_{k+1} = \cos \left( \frac{(m-k) \pi}{m} \right), \ k \in 0 \ldots m
	\]
	
	\noindent Для $m=4$ многочлен Чебышёва выглядит вот так:
	\[
	T_{4}(x) = \cos( 4 \arccos(x)) = 8x^4 - 8x^2 + 1
	\]
	
	\noindent А его точки альтернанса:
	\begin{align*}
		x_1 &= \cos \left( \frac{4\pi}{4} \right) = -1\\
		x_2 &= \cos \left( \frac{3\pi}{4} \right) = -\frac{\sqrt{2}}{2}\\
		x_3 &= \cos \left( \frac{2\pi}{4} \right) = 0\\
		x_4 &= \cos \left( \frac{1\pi}{4} \right) = \frac{\sqrt{2}}{2}\\
		x_5 &= \cos \left( \frac{0\pi}{4} \right) = 1\\
	\end{align*}

			(-1)
			\left(-1-\frac{\sqrt{2}}{2}\right)

		}
		\\
		&=
		\left(x+\frac{\sqrt{2}}{2}\right)
		x
		\left(x-\frac{\sqrt{2}}{2}\right)

		x^4-x^3-\frac{1}{2}x^2+\frac{1}{2}x

	\end{align*}
	
	
	\begin{align*}

		&=
		\frac{
			(x+1)
			x
			\left(x-\frac{\sqrt{2}}{2}\right)

		\\

	\begin{align*}

			(x+1)

			(x+1)

			\left(1+\frac{\sqrt{2}}{2}\right)

			\left(\frac{\sqrt{2}}{2}\right)
			\left(-1+\frac{\sqrt{2}}{2}\right)
		}
		\\
		&=
		-2(x+1)
		\left(x+\frac{\sqrt{2}}{2}\right)
		x
		(x-1)
		\\
		&=
		-2x^4-\sqrt{2}x^3+2x^2+\sqrt{2}x,
		\\[4mm]

	\noindent (\textbf{А теперь} $\mathbf{n = m}$)\\

	\[
	p_0(x) = \prod_{k=1}^{m} (x-x_k)
	\]

	\[A(p_0) = 0\]
	
	\noindent И, также 
	\[A_2(p_0) = 0\]
	
	\noindent Отсюда 
	\[A_1(p_0) = \lambda^* p_0'(x) =  0\]
	
	\noindent При фиксированном $x$ получаем условие на $x_k, k \in 1 \ldots m$ (лямбда не ноль)
	\[
	p_0'(x) = \frac{d}{dt} \left[\prod_{k=1}^{m} (t - x_k) \right]_{t=x} = \ 0
	\]
	
	\noindent Вспоминаем, что \eqref{eq:lagr_expansion}  $\forall p \in \mathbb{P}_m$ даёт нам
	\[
	p(t) = 
	\sum_{k=1}^{m} p(x_k) L_k(t) + 
	a_m(p) \prod_{k=1}^{m} (t-x_k) = 
	\sum_{k=1}^{m} p(x_k) L_k(t) + 
	a_m(p) p_0(t)
	\]
	
	\noindent Дифференцируя это равенство по $t$ получаем 
	\[
	p'(t) = 
	\sum_{k=1}^{m} p(x_k) L_k'(t) + 
	a_m(p) p_0'(t)
	\]
	
	\noindent Подставляя $t=x$ убиваем остаток
	\[
	p'(x) = 
	\sum_{k=1}^{m} p(x_k) L_k'(x)
	\]

	\begin{equation}
		\lambda^* p'(x) = 
		\sum_{k=1}^{m} p(x_k) ( \lambda^* L_k'(x) ) 
		\label{eq:re1}
	\end{equation}
	
	\noindent Но из условия \eqref{eq:cond1} 
	\[
	A(p) = 0, \quad \forall p \in \mathbb{P}_m
	\]
	
	\noindent То есть $\forall p \in \mathbb{P}_m$
	\begin{equation}
		\lambda^* p'(x) = 
		\sum_{k=1}^{m} p(x_k) ( a_k ) 
		\label{eq:re2}
	\end{equation}
	
	\noindent Из \eqref{eq:re1} и \eqref{eq:re2} получаем $\forall p \in \mathbb{P}_m$
	\[
	\sum_{k=1}^{m} p(x_k) ( a_k ) = \sum_{k=1}^{m} p(x_k) ( \lambda^* L_k'(x) ) 
	\] 
	
	\noindent Выбирая в качестве $p$ штуки которые равны $1$ и $0$ где надо, делаем вывод
	\[
	a_k = \lambda^* L_k'(x), \ \forall k \in 1 \ldots m 
	\]
	\[
	\omega_k (-1)^{k+1} = \lambda^* L_k'(x), \ \forall k \in 1 \ldots m 
	\]
	\[
	\omega_k  = (-1)^{k+1} \lambda^* L_k'(x), \ \forall k \in 1 \ldots m 
	\]

	\[
	1 = \sum_{k=1}^{m} \omega_k = \sum_{k=1}^{m} (-1)^{k+1} \lambda^* L_k'(x) = \lambda^* \sum_{k=1}^{m} (-1)^{k+1}  L_k'(x)
	\]
	
	\noindent получаем
	\[
	\lambda^* = \left( \sum_{k=1}^{m} (-1)^{k+1}  L_k'(x) \right) ^ {-1}
	\]
	
	\noindent финальное выражение для весов

	\omega_k  =

	}
	{
		\sum_{i=1}^{m} (-1)^{i+1}  L_i'(x)
	}, \ \forall k \in 1 \ldots m 
	\]
	
	\[
	L_k(x) = \underset{k \neq j}{\prod_{j=1}^{m}} \frac{x - x_j}{x_k - x_j}, \ k \in 1 \ldots m

	\subsection{А теперь ищем узлы}
	
	\subsection*{Если растяжка вправо}
	
	\noindent В таком сетапе считаем
	\[
	-1 = x_1 < x_2 < \ldots < x_m < 1 < z

	\tilde{L}_k(s), \
	\ k \in 1 \ldots m

	\[

	\ k \in 1 \ldots m}

	\noindent Так как
	\[
	L'_k(x) = L_k(x)\underset{i=1}{\sum_{i \neq k}^m} \frac{1}{x-x_i}
	\]
	\[
	\tilde{L}'_k(s) = \tilde{L}_k(s)\underset{i=1}{\sum_{i \neq k}^m} \frac{1}{s-t_{i-1}}

	\]

	\]
	
	\noindent Тогда ($ \forall i,j \ \ s_j \neq t_i $)
	\[
	\sum_{k=1}^{m} \frac{1}{s_i - t_{k-1}}  = 0, \ i=0,1,2
	\]
	
	\noindent Итого
	\[
	L'_k(x) = 
	\frac{1}{a} \tilde{L}'_k(s) =
	\frac{1}{a} \tilde{L}_k(s) \underset{i=1}{\sum_{i \neq k}^m} \frac{1}{s-t_{i-1}} =
	\frac{-\tilde{L}_k(s)}{a(s-t_{k-1})}
	\]

	\[
	\exists r \in \lbrace 1, \ \ldots, \ m-1 \rbrace : \quad s \in (t_{r-1}, t_r)
	\]

		sign \ L'_k(x) = (-1)^{k+r+1}
	\]

	sign \left( (-1)^{k+1} \lambda^* L_k'(x) \right) =
	sign (  \lambda^* )(-1)^r

	 Впрочем, если $r=1 \ (mod2)$ то $\lambda^*<0$ тоже $\omega_k > 0 \ \forall k \in 1 \ldots m$
	\[
	\lambda^* = \left( \sum_{k=1}^{m} (-1)^{k+1}  L_k'(x) \right) ^ {-1}
	\]

	\boxed{
		y^3-\frac7{16}y+\frac1{32}=0.
	}
	$$
	
	Для решения воспользуемся тригонометрической формой решения приведённого кубического уравнения
	
	$$
	y^3+py+q=0.
	$$
	
	В нашем случае
	
	$$
	p=-\frac7{16},
	\qquad
	q=\frac1{32}.
	$$
	
	Поскольку $p<0$, положим
	
	$$
	y=2\sqrt{-\frac p3}\cos\theta.
	$$
	
	Имеем
	
	$$
	2\sqrt{-\frac p3}
	=
	2\sqrt{\frac7{48}}
	=
	\sqrt{\frac7{12}},
	$$
	
	поэтому
	
	$$
	y=\sqrt{\frac7{12}}\cos\theta.
	$$
	
	После подстановки в уравнение $y^3+py+q=0$ и использования формулы
	
	$$
	4\cos^3\theta-3\cos\theta=\cos(3\theta)
	$$
	
	получаем
	
	$$
	\cos(3\theta)
	=
	\frac{3q}{2p}\sqrt{-\frac3p}.
	$$
	
	Вычислим правую часть:
	
	$$
	\frac{3q}{2p}
	=
	\frac{3(1/32)}
	{2(-7/16)}
	=
	-\frac3{28},
	$$
	
	а
	
	$$
	\sqrt{-\frac3p}
	=
	\sqrt{\frac{48}{7}}
	=
	4\sqrt{\frac37}.
	$$
	
	Следовательно,
	
	$$
	\cos(3\theta)
	=
	-\frac3{28}\,4\sqrt{\frac37}
	=
	-\frac3{7}\sqrt{\frac37}
	=
	-\frac{3\sqrt{21}}{49}.
	$$
	
	Таким образом,
	
	$$
	\boxed{
		\cos(3\theta)=-\frac{3\sqrt{21}}{49}.
	}
	$$
	
	Обозначим
	
	$$
	\varphi=
	\arccos\left(-\frac{3\sqrt{21}}{49}\right).
	$$
	
	Тогда три значения угла имеют вид
	
	$$
	\theta_k=\frac{\varphi+2\pi k}{3},
	\qquad
	k=0,1,2.
	$$
	
	Следовательно, три корня уравнения для $y$ равны
	
	$$
	y_k=
	\sqrt{\frac7{12}}
	\cos\left(
	\frac13
	\arccos\left(-\frac{3\sqrt{21}}{49}\right)
	+\frac{2\pi k}{3}
	\right),
	\qquad
	k=0,1,2.
	$$
	
	Возвращаясь к переменной $s$, поскольку
	
	$$
	s=y+\frac14,
	$$
	
	получаем
	
	$$
	\boxed{
		s_k=
		\frac14+
		\sqrt{\frac7{12}}
		\cos\left(
		\frac13
		\arccos\left(-\frac{3\sqrt{21}}{49}\right)
		+\frac{2\pi k}{3}
		\right),
		\qquad
		k=0,1,2.
	}
	$$
	
	Численные значения корней:
	
	$$
	\boxed{
		s_0 \approx 0.8723,
		\qquad
		s_1 \approx 0.3222,
		\qquad
		s_2 \approx -0.4446.
	}
	$$
	
	\noindent Выражаем параметры растяжения через $s$
	
	\begin{align*}
		s |_{t=x} &= \frac{x-b}{a} = \frac{x-b}{1-b}\\
		s (1-b) &= x-b\\
		b &= \frac{x-s}{1-s}\\
		a &= 1-b = 1-\frac{x-s}{1-s}
		= \frac{1-x}{1-s}
	\end{align*}
	
	\noindent Узлы получаем через
	\[
	x_k = a t_k + b =
	\frac{(1-x)t_k + x-s}{1-s} =
	\frac{ x(1 - t_k)  + (t_k - s)}{1-s}  ,
	\qquad
	k \in 1, \ldots m
	\]
	
	
	
	\subsection{Ограничение на $x$ от узлов (растяжка влево) }
	
	\noindent Вспоминаем геометрию узлов:
	\[
	z < -1 < x_1 < \ldots < x_{m-1} < x_m = 1
	\]
	
	\noindent Отсюда, с учетом растяжения влево ограничение на $x$:
	\[
	x_1 = x_1(x) \in (-1, \ t_1)
	\]
	
	
	\noindent Хотим растяжение с положительным коэффициентом
	\[
	a =  \frac{1-x}{1-s} > 0
	\]
	
	\noindent С учетом того что все корни $s < 1$ получаем условие
	\[
	x < 1
	\]
	
	\noindent Остается условие
	\begin{align*}
		-1 &< x_1 < t_1 \\
		-1 &< a t_1 + b < t_1 \\
		-1 &< \frac{1-x}{1-s} t_1 + \frac{x-s}{1-s} < t_1 \\
		s - 1 &< (1 - x) t_{1} + x - s < (1 - s)t_1 \\
		2s - 1 - t_1 &< x(1 - t_{1}) < s(1-t_1) \\ 
		\frac{2s - 1 - t_1}{(1 - t_{1})} &< x <  s
	\end{align*}
	
	\noindent Итого
	\[
	x \in \left(\frac{2s - 1 - t_1}{(1 - t_{1})}, \quad s  \right) = 
	\begin{cases}
		\left(\frac{2s_2 - 1 - t_1}{(1 - t_{1})}, \quad s_2  \right) \\
		\left(\frac{2s_1 - 1 - t_1}{(1 - t_{1})}, \quad s_1  \right) \\
		\left(\frac{2s_0 - 1 - t_1}{(1 - t_{1})}, \quad s_0  \right) \\
	\end{cases} \approx 
	\begin{cases}
		\left(-0.6924, \quad -0.4446  \right) \\
		\left(0.2059, \quad 0.3222 \right) \\
		\left(0.8503, \quad 0.8723 \right) \\
	\end{cases}
	\]
	
	\subsection{Ограничение на $x$ от весов (растяжка влево)}
	\noindent Хотим
	\[
	\omega_k  = (-1)^{k+1} \lambda^* L_k'(x) > 0 , \ \forall k \in 1 \ldots m 
	\]
	
	\noindent Хотим исследовать знаки $L'_k(x)$\\
	Вспоминаем что
	\[
	s = \frac{x-b}{a}
	\]
	
	\noindent и 
	\[
	x_k = a t_{k} + b, \ k \in 1 \ldots m
	\]
	
	\noindent Отсюда
	\[
	L_k(x) = 
	\underset{k \neq j}{\prod_{j=1}^{m}} \frac{x - x_j}{x_k - x_j} = 
	\underset{k \neq j}{\prod_{j=1}^{m}} \frac{s - t_{j}}{t_{k} - t_{j}} =
	\tilde{L}_k(s), \
	\ k \in 1 \ldots m
	\]
	
	\noindent То есть
	\[
	\boxed{
		L'_k(x) = \frac{1}{a} \tilde{L}'_k(s) , 
		\ k \in 1 \ldots m}
	\]
	
	\noindent Так как
	\[
	L'_k(x) = L_k(x)\sum_{i \neq k} \frac{1}{x-x_i}
	\]
	\[
	\tilde{L}'_k(s) = \tilde{L}_k(s)\sum_{i \neq k} \frac{1}{t-t_i}
	\]
	
	\noindent Помним что есть $q_0(s)$ 
	\[
	q_0(s) = \prod_{k=1}^{m} \left( s - t_{k} \right)
	\]
	
	\noindent A найденные выше $s_k$ это корни его производной
	\[
	q'_0(s) = q_0(s)\sum_{k=1}^{m} \frac{1}{s - t_{k}}  
	\]
	
	\noindent Тогда ($ \forall i,j \ \ s_j \neq t_i $)
	\[
	\sum_{k=1}^{m} \frac{1}{s_i - t_{k}}  = 0, \ i=0,1,2
	\]
	
	\noindent Итого
	\[
	L'_k(x) = 
	\frac{1}{a} \tilde{L}'_k(s) =
	\frac{1}{a} \tilde{L}_k(s) \underset{i=1}{\sum_{i \neq k}^m} \frac{1}{s-t_{i}} =
	\frac{-L_k(s)}{a(s-t_{k})}
	\]
	
	\noindent Ясно, что
	\[
	\exists r \in \lbrace 2, \ \ldots, \ m \rbrace : \quad s \in (t_{r-1}, t_r)
	\]
	
	\noindent Тогда ($a>0$)
	\[
	sign \ L'_k(x) = (-1)^{k+r}
	\]
	
	\[
	sign \ \omega_k  = 
	sign \left( (-1)^{k+1} \lambda^* L_k'(x) \right) =
	sign (  \lambda^* )(-1)^{r+1}
	\]
	
	\noindent Знак лямбды находим через\\
	Если $r=0 \ (mod2)$ то $\lambda^*<0$ и $\omega_k > 0 \ \forall k \in 1 \ldots m$.\\
	Впрочем, если $r=1 \ (mod2)$ то $\lambda^*>0$ и тоже $\omega_k > 0 \ \forall k \in 1 \ldots m$
	\[
	\lambda^* = \left( \sum_{k=1}^{m} (-1)^{k+1}  L_k'(x) \right) ^ {-1}
	\]

	\section{Решение системы}
	
	\subsection{Исходная система}
	
	Фиксируем $x\in\mathbb{R}$ и рассматриваем
	
	$$
	f(x)=
	\begin{pmatrix}
		1\\
		x\\
		x^2\\
		x^3\\
		x^4
	\end{pmatrix},
	\qquad
	c=f'(x)=
	\begin{pmatrix}
		0\\
		1\\
		2x\\
		3x^2\\
		4x^3
	\end{pmatrix}.
	$$
	
	Пусть
	
	$$
	\xi=
	\begin{pmatrix}
		x_1&x_2&x_3&x_4\\
		\omega_1&\omega_2&\omega_3&\omega_4
	\end{pmatrix}.
	$$
	
	При $m=n=4$ имеем
	
	$$
	\lambda^*
	\begin{pmatrix}
		0\\
		1\\
		2x\\
		3x^2\\
		4x^3
	\end{pmatrix}
	=
	\sum_{k=1}^4
	(-1)^{k+1}\omega_k
	\begin{pmatrix}
		1\\
		x_k\\
		x_k^2\\
		x_k^3\\
		x_k^4
	\end{pmatrix}.
	$$
	
	Фиксируем крайние узлы
	
	$$
	x_1=-1,
	\qquad
	x_4=1.
	$$
	
	Для удобства обозначим
	
	$$
	x_2=u,
	\qquad
	x_3=v.
	$$
	
	Таким образом,
	
	$$
	x_1=-1,\qquad
	x_2=u,\qquad
	x_3=v,\qquad
	x_4=1.
	$$
	
	\subsection{Система по компонентам}
	
	После покомпонентного раскрытия получаем
	
	$$
	\begin{cases}
		\omega_1-\omega_2+\omega_3-\omega_4=0,\\[2mm]
		-\omega_1-\omega_2u+\omega_3v-\omega_4=\lambda^*,\\[2mm]
		\omega_1-\omega_2u^2+\omega_3v^2-\omega_4=2\lambda^*x,\\[2mm]
		-\omega_1-\omega_2u^3+\omega_3v^3-\omega_4=3\lambda^*x^2,\\[2mm]
		\omega_1-\omega_2u^4+\omega_3v^4-\omega_4=4\lambda^*x^3.
	\end{cases}
	$$
	
	Дополнительно выполняется условие нормировки
	
	$$
	\omega_1+\omega_2+\omega_3+\omega_4=1.
	$$
	
	\subsection{Выражение $\omega_3$ и $\omega_4$}
	
	Из первого уравнения системы имеем
	
	$$
	\omega_3-\omega_4=\omega_2-\omega_1.
	$$
	
	Из условия нормировки
	
	$$
	\omega_3+\omega_4=1-\omega_1-\omega_2.
	$$
	
	Складывая и вычитая эти равенства, получаем
	
	$$
	\boxed{
		\omega_3=\frac12-\omega_1
	}
	$$
	
	и
	
	$$
	\boxed{
		\omega_4=\frac12-\omega_2.
	}
	$$
	
	Следовательно,
	
	$$
	\omega_1=\omega_1,\qquad
	\omega_2=\omega_2,\qquad
	\omega_3=\frac12-\omega_1,\qquad
	\omega_4=\frac12-\omega_2.
	$$
	
	\subsection{Исключение $\lambda^*$}
	
	Введём обозначение
	
	$$
	M_r=
	(-1)^r\omega_1
	-\omega_2u^r
	+\left(\frac12-\omega_1\right)v^r
	-\left(\frac12-\omega_2\right).
	$$
	
	Тогда четыре неоднородных уравнения исходной системы принимают вид
	
	$$
	M_1=\lambda^*,
	$$
	
	$$
	M_2=2\lambda^*x,
	$$
	
	$$
	M_3=3\lambda^*x^2,
	$$
	
	$$
	M_4=4\lambda^*x^3.
	$$
	
	Из первого равенства
	
	$$
	\lambda^*=M_1.
	$$
	
	Поэтому вместо остальных трёх уравнений можно рассматривать
	
	$$
	\boxed{
		F_1=M_2-2xM_1=0,
	}
	$$
	
	$$
	\boxed{
		F_2=M_3-3x^2M_1=0,
	}
	$$
	
	$$
	\boxed{
		F_3=M_4-4x^3M_1=0.
	}
	$$
	
	%check
	После раскрытия скобок получаем
	
	$$
	\boxed{
		\begin{aligned}
			F_1={}&
			\omega_1(1+2x-v^2+2vx)
			\
			&+\omega_2(-u^2+2ux-2x+1)
			\
			&+\frac{v^2}{2}-vx+x-\frac12
			=0,
		\end{aligned}
	}
	$$
	
	$$
	\boxed{
		\begin{aligned}
			F_2={}&
			\omega_1(3x^2-1-v^3+3vx^2)
			\
			&+\omega_2(-u^3+3ux^2-3x^2+1)
			\
			&+\frac{v^3}{2}
			-\frac32vx^2
			+\frac32x^2-\frac12
			=0,
		\end{aligned}
	}
	$$
	
	$$
	\boxed{
		\begin{aligned}
			F_3={}&
			\omega_1(1+4x^3-v^4+4vx^3)
			\
			&+\omega_2(-u^4+4ux^3-4x^3+1)
			\
			&+\frac{v^4}{2}
			-2vx^3
			+2x^3-\frac12
			=0.
		\end{aligned}
	}
	$$
	
	\subsection{Базисные многочлены Лагранжа}

	$$

	\subsection{Введение $A$, $B$, $C$}
	
	Обозначим

	}
	$$
	
	$$
	\boxed{
		B=

	$$

	Тогда уравнение для $u=x_2$ принимает вид

	\boxed{

	$$
	
	При условии
	
	$$
	x\ne0
	$$
	
	имеем
	
	$$
	\boxed{
		x_2=u=
		\frac{-B\pm\sqrt{B^2-4AC}}{2A}.
	}
	$$
	
	\noindent Нам очень повезло с дискриминантом
	\[
	\begin{aligned}
		\Delta
		&=B^2-4AC\\
		&=\left(2\alpha x-6x^2+2\right)^2
		-4(2x)\left(-3\alpha x^2+\alpha+4x^3-2x\right)
		\\[2mm]
		&=4\alpha^2x^2+4x^4-8x^2+4
		\\[2mm]
		&=4\left(\alpha^2x^2+x^4-2x^2+1\right)
		\\[2mm]
		&=4\left(\alpha^2x^2+(x^2-1)^2\right) > 0
	\end{aligned}
	\]
	

	
	После этого
	
	$$
	\boxed{
		x_3=x_2+\alpha.
	}
	$$
	
	

	
	\subsection{Итоговая схема решения}
	
	Таким образом,
	
	$$
	\boxed{
		x_1=-1,\qquad
		x_4=1,\qquad
		x_3-x_2=\alpha.
	}
	$$
	
	Полагая
	
	$$
	x_2=u,\qquad
	x_3=u+\alpha,
	$$
	
	получаем
	
	$$
	\boxed{
		x_2=u,\qquad
		x_3=u+\alpha.
	}
	$$
	
	Значение $u$ определяется из квадратного уравнения
	
	$$
	\boxed{
		2xu^2
		+
		\left(
		2\alpha x-6x^2+2
		\right)u
		+
		\left(
		-3\alpha x^2+\alpha+4x^3-2x
		\right)
		=0.
	}
	$$
	
	При $2x\ne0$
	
	$$
	\boxed{
		x_2=
		\frac{-B\pm\sqrt{B^2-4AC}}{2A},
		\qquad
		x_3=x_2+\alpha,
	}
	$$
	
	где
	
	$$
	\boxed{
		A=2x,
	}
	$$
	
	$$
	\boxed{
		B=
		2\alpha x-6x^2+2,
	}
	$$
	
	$$
	\boxed{
		C=
		-3\alpha x^2+\alpha+4x^3-2x.
	}
	$$
	
	После определения узлов вычисляются веса:
	
	$$
	\boxed{
		\omega_k=
		\frac{
			(-1)^{k+1}L_k'(x)
		}{
			L_1'(x)-L_2'(x)+L_3'(x)-L_4'(x)
		}.
	}
	$$
	
	Наконец,
	
	$$
	\boxed{
		\lambda^*
		=
		-\omega_1-\omega_2x_2+\omega_3x_3-\omega_4.
	}
	$$
	
	При выборе ветви $\pm$ необходимо учитывать допустимость узлов
	
	$$
	\boxed{
		-1<x_2<x_3<1
	}
	$$
	
	\newpage
	\subsection{Явный выбор параметра $\alpha$}
	
	
	Так как
	
	$$
	x_2=u,\qquad x_3=u+\alpha
	$$
	
	и $\alpha>0$, неравенство $x_2<x_3$ выполняется автоматически. Поэтому достаточно проверить
	
	$$
	u>-1
	$$
	
	и
	
	$$
	u+\alpha<1,
	$$
	
	то есть
	
	$$
	\boxed{
		-1<u<1-\alpha.
	}
	$$
	
	Покажем, что этого можно добиться явным выбором
	
	$$
	\boxed{
		\alpha=\frac{1-|x|}{2}.
	}
	$$
	
	Поскольку $-1<x<1$, имеем
	
	$$
	\boxed{
		0<\alpha<\frac12.
	}
	$$
	
	
	\subsubsection{Случай $x>0$}
	
	Положим
	
	$$
	t=x,\qquad 0<t<1.
	$$
	
	Тогда
	
	$$
	\alpha=\frac{1-t}{2}.
	$$
	
	Выберем плюсовую ветвь:
	
	$$
	u=u_+
	=
	\frac{
		-\alpha t+3t^2-1
		+
		\sqrt{\alpha^2t^2+(1-t^2)^2}
	}{2t}.
	$$
	
	Подставляя
	
	$$
	\alpha=\frac{1-t}{2},
	$$
	
	получаем
	
	$$
	\alpha^2t^2
	=
	\frac{t^2(1-t)^2}{4}.
	$$
	
	Поэтому
	\begin{align*}
		\alpha^2t^2+(1-t^2)^2
		&=
		\frac{t^2(1-t)^2}{4}
		+(1-t^2)^2\
		&=
		\frac14
		\left(
		t^2(1-t)^2+4(1-t^2)^2
		\right).
	\end{align*}

	
	Для проверки расположения корня удобнее оставить радикал в исходном виде. После подстановки $\alpha=(1-t)/2$ имеем
	
	$$
	u
	=
	\frac{
		-\frac{t(1-t)}2+3t^2-1
		+
		\sqrt{
			\frac{t^2(1-t)^2}{4}+(1-t^2)^2
		}
	}{2t}.
	$$
	
	После вынесения множителя $1/2$ из числителя:
	
	$$
	\boxed{
		u=
		\frac{
			7t^2-t-2+
			\sqrt{
				t^2(1-t)^2+4(1-t^2)^2
			}
		}{4t}.
	}
	$$
	
	Следовательно, необходимо проверить
	
	$$
	u>-1
	$$
	
	и
	
	$$
	u<1-\alpha=\frac{1+t}{2}.
	$$
	
	Первое неравенство эквивалентно
	
	$$
	\frac{
		7t^2-t-2+
		\sqrt{
			t^2(1-t)^2+4(1-t^2)^2
		}
	}{4t}>-1.
	$$
	
	Поскольку $4t>0$, это равносильно
	
	$$
	7t^2-t-2+
	\sqrt{
		t^2(1-t)^2+4(1-t^2)^2
	}
	>-4t,
	$$
	
	то есть
	
	$$
	7t^2+3t-2+
	\sqrt{
		t^2(1-t)^2+4(1-t^2)^2
	}>0.
	$$
	
	Второе неравенство
	
	$$
	u<\frac{1+t}{2}
	$$
	
	равносильно
	
	$$
	7t^2-t-2+
	\sqrt{
		t^2(1-t)^2+4(1-t^2)^2
	}
	<
	2t(1+t),
	$$
	
	то есть
	
	$$
	5t^2-3t-2+
	\sqrt{
		t^2(1-t)^2+4(1-t^2)^2
	}
	<0.
	$$
	
	Таким образом, для $0<t<1$ проверка сводится к двум неравенствам
	
	$$
	\boxed{
		7t^2+3t-2+
		\sqrt{
			t^2(1-t)^2+4(1-t^2)^2
		}>0
	}
	$$
	
	и
	
	$$
	\boxed{
		5t^2-3t-2+
		\sqrt{
			t^2(1-t)^2+4(1-t^2)^2
		}<0.
	}
	$$
	
	Для первого неравенства достаточно заметить, что
	
	$$
	\sqrt{
		t^2(1-t)^2+4(1-t^2)^2
	}
	\geq
	2(1-t^2),
	$$
	
	поскольку $0<t<1$. Поэтому
	\begin{align*}
		&7t^2+3t-2+
		\sqrt{
			t^2(1-t)^2+4(1-t^2)^2
		}\
		&\qquad\geq
		7t^2+3t-2+2(1-t^2)\
		&\qquad=
		5t^2+3t\
		&\qquad>0.
	\end{align*}
	Следовательно,
	
	$$
	u>-1.
	$$
	
	Для второго неравенства необходимо доказать
	
	$$
	\sqrt{
		t^2(1-t)^2+4(1-t^2)^2
	}
	<
	2+3t-5t^2.
	$$
	
	Правая часть положительна при $0<t<1$, поскольку
	
	$$
	2+3t-5t^2
	=
	(1-t)(5t+2)>0.
	$$
	
	Делим обе части на $(1-t)$. Получаем
	
	\[
	\sqrt{t^2+4} < (5t+2)
	\]
	
	или,
	\[
		t^2 + 4 < 25t^2 + 20t + 4
	\]
	
	FINALLY,
	\[
	24t^2 + 20t > 0 
	\]
	
	и потому
	
	$$
	u<\frac{1+t}{2}=1-\alpha.
	$$
	
	Таким образом, при $0<x<1$ имеем
	
	$$
	\boxed{
		-1<u<u+\alpha<1.
	}
	$$
	
	\subsubsection{Случай $x<0$}
	Проверка такая же как и в случае с $x<0$ показывает, что
	\[
	\alpha = \frac{1-|x|}{2}
	\]
	подходит под все условия
	
	\subsubsection{Случай $x=0$}
	
	При $x=0$ коэффициент при $u^2$ равен
	
	$$
	A=2x=0,
	$$
	
	поэтому квадратное уравнение вырождается в линейное.
	
	Подставляя $x=0$ в исходное уравнение, получаем
	
	$$
	2u+\alpha=0.
	$$
	
	Следовательно,
	
	$$
	u=-\frac{\alpha}{2}.
	$$
	
	Для выбранной формулы
	
	$$
	\alpha=\frac{1-|x|}{2}
	$$
	
	при $x=0$ имеем
	
	$$
	\alpha=\frac12.
	$$
	
	Поэтому
	
	$$
	x_2=u=-\frac14,
	$$
	
	а
	
	$$
	x_3=u+\alpha
	=
	-\frac14+\frac12
	=
	\frac14.
	$$
	
	Таким образом,
	
	$$
	\boxed{
		-1<-\frac14<\frac14<1.
	}
	$$
	
	\subsection{Итог}
	
	Итак, для каждого
	
	$$
	\boxed{x\in(-1,1)}
	$$
	
	можно выбрать параметр
	
	$$
	\boxed{
		\alpha=\frac{1-|x|}{2}.
	}
	$$
	
	При $x\neq0$ берём плюсовую ветвь
	
	$$
	\boxed{
		x_2=
		\frac{
			-\alpha x+3x^2-1+
			\sqrt{\alpha^2x^2+(1-x^2)^2}
		}{2x},
	}
	$$
	
	а затем полагаем
	
	$$
	\boxed{
		x_3=x_2+\alpha.
	}
	$$
	
	При $x=0$ имеем
	
	$$
	\boxed{
		x_2=-\frac14,
		\qquad
		x_3=\frac14.
	}
	$$
	
	Поскольку
	
	$$
	0<\alpha\leq\frac12
	$$
	
	и выполнено
	
	$$
	-1<x_2<x_3<1,
	$$
		
	\subsection{Условие на $x$ от весов (два на границе)}
	
	
	имеем узлы
	\[
	x_1=-1,\qquad x_2=u,\qquad x_3=v,\qquad x_4=1,
	\]
	причём \(v=u+\alpha\), а параметр \(\alpha\) выбирается как
	\[
	\alpha=\frac{1-|x|}{2},\qquad x\in(-1,1).
	\]
	Веса даются формулой
	\[
	\omega_k=\frac{(-1)^{k+1}L_k'(x)}{D},\qquad
	D=L_1'(x)-L_2'(x)+L_3'(x)-L_4'(x).
	\]
	Нужно доказать, что \(\omega_k>0\) для всех \(k=1,\dots,4\). Достаточно показать, что все величины \((-1)^{k+1}L_k'(x)\) имеют один и тот же знак, а именно положительный.
	
	\subsubsection*{Вспоминаем, что \(p'(x)=0\)}
	
	Введём многочлен
	\[
	p(t)=(t+1)(t-u)(t-v)(t-1).
	\]
	
	\noindent Тогда
	\[
	p'(x)=0.
	\]

	\subsubsection*{2. Производные базисных многочленов Лагранжа при \(p'(x)=0\)}
	
	%CHECK
	Для базисных многочленов Лагранжа (проверить в лоб)
	\[
	L_k(t)=\frac{p(t)}{(t-x_k)p'(x_k)}
	\]
	при \(p'(x)=0\) получаем
	\[
	L_k'(x)=-\frac{p(x)}{(x-x_k)^2p'(x_k)}.
	\]
	Корни упорядочены:
	\[
	-1<u<v<1.
	\]
	Тогда знаки \(p'(x_k)\) чередуются:
	\[
	\operatorname{sign}p'(x_k)=(-1)^k,\qquad k=1,2,3,4.
	\]
	Следовательно,
	\[
	\operatorname{sign}L_k'(x)
	=
	-\operatorname{sign}p(x)\cdot(-1)^k
	=
	\operatorname{sign}p(x)\cdot(-1)^{k+1}.
	\]
	Поэтому
	\[
	\operatorname{sign}\bigl((-1)^{k+1}L_k'(x)\bigr)=\operatorname{sign}p(x)
	\]
	для всех \(k=1,\dots,4\). Значит, все \((-1)^{k+1}L_k'(x)\) имеют одинаковый знак, совпадающий со знаком \(p(x)\).
	
	\subsubsection*{3. Покажем, что \(p(x)>0\)}
	
	
	%CHECK
	При выбранном \(\alpha=\frac{1-|x|}{2}\) для \(x>0\) положим \(t=x\). Из явной формулы для \(u\) можно доказать, что
	\[
	u<t<v=u+\alpha.
	\]
	
	
	\noindent Итак, \(x\in(u,v)\). Тогда
	\[
	p(x)=(x+1)(x-u)(x-v)(x-1) > 0, \quad \forall x \in (-1, \ 1)
	\]
	При \(x\in(-1,1)\) имеем \(x+1>0\), \(x-1<0\). А так как \(x\in(u,v)\), то \(x-u>0\), \(x-v<0\). Произведение двух положительных и двух отрицательных множителей положительно:
	\[
	p(x)>0.
	\]
	
	\subsubsection*{4. Вывод}
	
	Мы получили, что
	\[
	(-1)^{k+1}L_k'(x)>0\qquad \forall k=1,\dots,4.
	\]
	Следовательно,
	\[
	D=\sum_{k=1}^4(-1)^{k+1}L_k'(x)>0,
	\]
	и поэтому
	\[
	\omega_k=\frac{(-1)^{k+1}L_k'(x)}{D}>0\qquad \forall k=1,\dots,4.
	\]
	
	Таким образом, при выборе \(\alpha=\frac{1-|x|}{2}\) все веса \(\omega_k\) положительны для любого \(x\in(-1,1)\).
