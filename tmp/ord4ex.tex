	\section{Обозначения}
	Фиксируем $x \in \mathbb{R}$\\
	Пусть $f(x) = (1, x, \ldots, x^m)^\mathrm{T}$, $c=f'(x)$\\
	
	\noindent Рассмотрим план
	\[
	\xi = 
	\left(
	\begin{matrix}
		x_1 \ x_2 \ldots x_n \\
		\omega_1 \ \omega_2 \ldots \omega_n	
	\end{matrix}
	\right)
	\]
	
	\noindent Та самая система для плана с $n$ узлами имеет вид:
	\[
	c = h M(\xi) p = h \sum_{k=1}^{n} \omega_k <p,f(x_k)> f(x_k)
	\]	
	
	\noindent или, c учётом $h = (\lambda ^*)^{-1}$ и чередования знаков т.е. $<p,f(x_k)> = (-1)^{k+1}$
	\[
	\lambda ^*
	\begin{pmatrix}
		0 \\
		1 \\
		\vdots \\
		m x^{m-1}
	\end{pmatrix} = 
	\sum_{k=1}^{n} a_k 
	\begin{pmatrix}
		1 \\
		x_k \\
		\vdots \\
		x_k^{m}
	\end{pmatrix}, \quad a_k = \omega_k <p,f(x_k)>
	\]
	
	\noindent Введём линейные операторы $A_1, A_2, A: \mathbb{P}_m \to \mathbb{R}$:
	\[
	A_1(p) := \lambda^* p'(x)
	\]
	\[
	A_2(p) := \sum_{k=1}^{n} a_k p(x_k)
	\]
	\[
	A(p) := A_1(p) - A_2(p)
	\]
	
	\noindent Нетрудно видеть, что наша система эквивалентна вот такому:
	\[
	\lbrace 1, x, \ldots, x^m \rbrace \subset \ker A
	\]
	
	\noindent Т.к $A$ линейно, то это в свою очередь эквивалентно
	\begin{equation}
		\mathbb{P}_m \subset \ker A
		\label{eq:cond1}
	\end{equation}

	\section{Чебышёвские планы}
	\subsection{Ищем веса в чебышёвских планах}
	
	
	\noindent (\textbf{в этом разделе} $\mathbf{n = m+1}$)\\ 
	
	
	\noindent Вспоминаем, что \eqref{eq:lagr_expansion}  $\forall p \in \mathbb{P}_m$ даёт нам
	\[
	p(t) = 
	\sum_{k=1}^{m+1} p(x_k) L_k(t), \  \forall t \in \mathbb{R}
	\]
	
	\noindent Дифференцируя это равенство по $t$ и подставляя $t=x$ получаем 
	\[
	p'(x) = 
	\sum_{k=1}^{m+1} p(x_k) L_k'(x)
	\]
	
	
	\noindent Получили, что $\forall p \in \mathbb{P}_m$
	\begin{equation}
		\lambda^* p'(x) = 
		\sum_{k=1}^{m+1} p(x_k) ( \lambda^* L_k'(x) ) 
		\label{eq:re1_2}
	\end{equation}
	
	\noindent Но из условия \eqref{eq:cond1} 
	\[
	A(p) = 0, \quad \forall p \in \mathbb{P}_m
	\]
	
	\noindent То есть $\forall p \in \mathbb{P}_m$
	\begin{equation}
		\lambda^* p'(x) = 
		\sum_{k=1}^{m+1} p(x_k) ( a_k ) 
		\label{eq:re2_2}
	\end{equation}
	
	\noindent Из \eqref{eq:re1_2} и \eqref{eq:re2_2} получаем $\forall p \in \mathbb{P}_m$
	\[
	\sum_{k=1}^{m+1} p(x_k) ( a_k ) = \sum_{k=1}^{m+1} p(x_k) ( \lambda^* L_k'(x) ) 
	\] 
	
	\noindent Выбирая в качестве $p$ штуки которые равны $1$ и $0$ где надо ($L_k(x) ???$), делаем вывод
	\[
	a_k = \lambda^* L_k'(x), \ \forall k \in 1 \ldots m+1 
	\]
	\[
	\omega_k (-1)^{k+1} = \lambda^* L_k'(x), \ \forall k \in 1 \ldots m+1 
	\]
	\[
	\omega_k  = (-1)^{k+1} \lambda^* L_k'(x), \ \forall k \in 1 \ldots m+1 
	\]
	
	
	\noindent Юзая условие на веса $\sum_{k=1}^{m+1} \omega_k = 1$ ищем лямбду:
	\[
	1 = \sum_{k=1}^{m+1} \omega_k = \sum_{k=1}^{m+1} (-1)^{k+1} \lambda^* L_k'(x) = \lambda^* \sum_{k=1}^{m+1} (-1)^{k+1}  L_k'(x)
	\]
	
	\noindent получаем
	\[
	\lambda^* = \left( \sum_{k=1}^{m+1} (-1)^{k+1}  L_k'(x) \right) ^ {-1}
	\]
	
	\noindent финальное выражение для весов
	\[
	\omega_k  = 
	\frac
	{
		(-1)^{k+1} L_k'(x)
	}
	{
		\sum_{i=1}^{m+1} (-1)^{i+1}  L_i'(x)
	}, \ \forall k \in 1 \ldots m+1 
	\]
	
	\[
	L_k(x) = \underset{k \neq j}{\prod_{j=1}^{m+1}} \frac{x - x_j}{x_k - x_j}, \ k \in 1 \ldots m+1
	\]
	
	\subsection{Ограничение на $x$ от весов}
	
	Мы требуем $\forall k \in 1 \ldots m+1 \ \omega_k > 0 $
	
	\noindent То есть нужно решить систему неравенств относительно $x$:
	\[
	\omega_k  = (-1)^{k+1} \lambda^* L_k'(x)  > 0 , \ \forall k \in 1 \ldots m+1 
	\]
	
	
	
	\subsection{Ищем узлы в чебышёвских планах}
	
	\noindent А узлы это просто точки альтернанса полинома Чебышёва 1-го рода 
	\[
	T_{m}(x) = \cos( m \arccos(x))
	\]
	\noindent Они имеют вид 
	\[
	x_{k+1} = \cos \left( \frac{(m-k) \pi}{m} \right), \ k \in 0 \ldots m
	\]
	
	\noindent Для $m=4$ многочлен Чебышёва выглядит вот так:
	\[
	T_{4}(x) = \cos( 4 \arccos(x)) = 8x^4 - 8x^2 + 1
	\]
	
	\noindent А его точки альтернанса:
	\begin{align*}
		x_1 &= \cos \left( \frac{4\pi}{4} \right) = -1\\
		x_2 &= \cos \left( \frac{3\pi}{4} \right) = -\frac{\sqrt{2}}{2}\\
		x_3 &= \cos \left( \frac{2\pi}{4} \right) = 0\\
		x_4 &= \cos \left( \frac{1\pi}{4} \right) = \frac{\sqrt{2}}{2}\\
		x_5 &= \cos \left( \frac{0\pi}{4} \right) = 1\\
	\end{align*}
		
		
	\noindent Теперь найдем веса, для этого сначала выпишем базисные многочлены и их производные
	
	\begin{align*}
		L_1(x)
		&=
		\frac{
			\left(x+\frac{\sqrt{2}}{2}\right)
			x
			\left(x-\frac{\sqrt{2}}{2}\right)
			(x-1)
		}{
			\left(-1+\frac{\sqrt{2}}{2}\right)
			(-1)
			\left(-1-\frac{\sqrt{2}}{2}\right)
			(-2)
		}
		\\
		&=
		\left(x+\frac{\sqrt{2}}{2}\right)
		x
		\left(x-\frac{\sqrt{2}}{2}\right)
		(x-1)
		\\
		&=
		x^4-x^3-\frac{1}{2}x^2+\frac{1}{2}x
		\\[4mm]
		L'_1(x) &= 4x^3 -3x^2-x+\frac{1}{2}
	\end{align*}
	
	
	\begin{align*}
		L_2(x)
		&=
		\frac{
			(x+1)
			x
			\left(x-\frac{\sqrt{2}}{2}\right)
			(x-1)
		}{
			\left(-\frac{\sqrt{2}}{2}+1\right)
			\left(-\frac{\sqrt{2}}{2}\right)
			(-\sqrt{2})
			\left(-\frac{\sqrt{2}}{2}-1\right)
		}
		\\
		&=
		-2(x+1)
		x
		\left(x-\frac{\sqrt{2}}{2}\right)
		(x-1)
		\\
		&=
		-2x^4+\sqrt{2}x^3+2x^2-\sqrt{2}x \\[4mm]
		L'_2(x) &= -8x^3 + 3\sqrt{2}x^2 + 4x - \sqrt{2} 	
	\end{align*}
		
	\begin{align*}
		L_3(x)
		&=
		\frac{
			(x+1)
			\left(x+\frac{\sqrt{2}}{2}\right)
			\left(x-\frac{\sqrt{2}}{2}\right)
			(x-1)
		}{
			(1)
			\left(\frac{\sqrt{2}}{2}\right)
			\left(-\frac{\sqrt{2}}{2}\right)
			(-1)
		}
		\\
		&=
		2(x+1)
		\left(x+\frac{\sqrt{2}}{2}\right)
		\left(x-\frac{\sqrt{2}}{2}\right)
		(x-1)
		\\
		&=
		2x^4-3x^2+1,
		\\[4mm]
		L'_3(x) &= 8x^3 - 6x
	\end{align*}
	
	\begin{align*}
		L_4(x)
		&=
		\frac{
			(x+1)
			\left(x+\frac{\sqrt{2}}{2}\right)
			x
			(x-1)
		}{
			\left(1+\frac{\sqrt{2}}{2}\right)
			\sqrt{2}
			\left(\frac{\sqrt{2}}{2}\right)
			\left(-1+\frac{\sqrt{2}}{2}\right)
		}
		\\
		&=
		-2(x+1)
		\left(x+\frac{\sqrt{2}}{2}\right)
		x
		(x-1)
		\\
		&=
		-2x^4-\sqrt{2}x^3+2x^2+\sqrt{2}x,
		\\[4mm]
		L'_4(x) &= -8x^3 - 3\sqrt{2}x^2 + 4x + \sqrt{2}
	\end{align*}
		
	\begin{align*}
		L_5(x)
		&=
		\frac{
			(x+1)
			\left(x+\frac{\sqrt{2}}{2}\right)
			x
			\left(x-\frac{\sqrt{2}}{2}\right)
		}{
			(2)
			\left(1+\frac{\sqrt{2}}{2}\right)
			(1)
			\left(1-\frac{\sqrt{2}}{2}\right)
		}
		\\
		&=
		(x+1)
		\left(x+\frac{\sqrt{2}}{2}\right)
		x
		\left(x-\frac{\sqrt{2}}{2}\right)
		\\
		&=
		x^4+x^3-\frac{1}{2}x^2-\frac{1}{2}x\\[4mm]
		L'_5(x) &= 4x^3 + 3x^2 -x - \frac{1}{2}
	\end{align*}
	
	
	
	\begin{align*}
		\sum_{k=1}^{5}(-1)^{k+1}L_k'(x)
		&=
		L_1'(x)-L_2'(x)+L_3'(x)-L_4'(x)+L_5'(x) = \\
		&=
		(4x^3 -3x^2-x+\frac{1}{2})\\
		&
		- (-8x^3 + 3\sqrt{2}x^2 + 4x - \sqrt{2})\\
		&
		+ (8x^3 - 6x)\\
		&
		-(-8x^3 - 3\sqrt{2}x^2 + 4x + \sqrt{2})\\
		&
		+(4x^3 + 3x^2 -x - \frac{1}{2}) =\\
		& =
		32x^3-16x \\
		& =
		16x(2x^2-1)
	\end{align*}
	
	\noindent Отсюда лямбда равна
	\[
	\lambda^* = 
	\left( \sum_{k=1}^{m+1} (-1)^{k+1}  L_k'(x) \right) ^ {-1} = 
	\frac{1}{16x(2x^2-1)}
	\]
	
	\noindent Теперь находим веса из формулы
	\[\omega_k  = (-1)^{k+1} \lambda^* L_k'(x), \ \forall k \in 1 \ldots m+1 \]
	\begin{align*}
		\omega_1 &= 
		(-1)^{1+1} \lambda^* L_1'(x) = 
		\frac{(4x^3 -3x^2-x+\frac{1}{2})}{16x(2x^2-1)}\\[3mm]
		\omega_2 &= 
		(-1)^{2+1} \lambda^* L_2'(x) = 
		\frac{- (-8x^3 + 3\sqrt{2}x^2 + 4x - \sqrt{2})}{16x(2x^2-1)}\\[3mm]
		\omega_3 &= 
		(-1)^{3+1} \lambda^* L_3'(x) = 
		\frac{(8x^3 - 6x)}{16x(2x^2-1)}\\[3mm]
		\omega_4 &= 
		(-1)^{4+1} \lambda^* L_4'(x) = 
		\frac{-(-8x^3 - 3\sqrt{2}x^2 + 4x + \sqrt{2})}{16x(2x^2-1)}\\[3mm]
		\omega_5 &= 
		(-1)^{5+1} \lambda^* L_5'(x) = 
		\frac{(4x^3 + 3x^2 -x - \frac{1}{2})}{16x(2x^2-1)}\\
	\end{align*}
	
	\noindent В конце концов получаем что план имеет вид
	
	\[
	\begin{pmatrix}
	-1 & -\frac{\sqrt2}{2} & 0 & \frac{\sqrt2}{2} & 1 \\
	\frac{(4x^3 -3x^2-x+\frac{1}{2})}{16x(2x^2-1)} &
	\frac{- (-8x^3 + 3\sqrt{2}x^2 + 4x - \sqrt{2})}{16x(2x^2-1)} &
	\frac{(8x^3 - 6x)}{16x(2x^2-1)} &
	\frac{-(-8x^3 - 3\sqrt{2}x^2 + 4x + \sqrt{2})}{16x(2x^2-1)} &
	\frac{(4x^3 + 3x^2 -x - \frac{1}{2})}{16x(2x^2-1)}
	\end{pmatrix}
	\]

	\subsection{Ищем веса в получебышёвских планах}
	\noindent (\textbf{А теперь} $\mathbf{n = m}$)\\ 
	Теперь рассмотрим 
	\[
	p_0(x) = \prod_{k=1}^{m} (x-x_k)
	\]
	
	\noindent Для него
	\[A(p_0) = 0\]
	
	\noindent И, также 
	\[A_2(p_0) = 0\]
	
	\noindent Отсюда 
	\[A_1(p_0) = \lambda^* p_0'(x) =  0\]
	
	\noindent При фиксированном $x$ получаем условие на $x_k, k \in 1 \ldots m$ (лямбда не ноль)
	\[
	p_0'(x) = \frac{d}{dt} \left[\prod_{k=1}^{m} (t - x_k) \right]_{t=x} = \ 0
	\]
	
	\noindent Вспоминаем, что \eqref{eq:lagr_expansion}  $\forall p \in \mathbb{P}_m$ даёт нам
	\[
	p(t) = 
	\sum_{k=1}^{m} p(x_k) L_k(t) + 
	a_m(p) \prod_{k=1}^{m} (t-x_k) = 
	\sum_{k=1}^{m} p(x_k) L_k(t) + 
	a_m(p) p_0(t)
	\]
	
	\noindent Дифференцируя это равенство по $t$ получаем 
	\[
	p'(t) = 
	\sum_{k=1}^{m} p(x_k) L_k'(t) + 
	a_m(p) p_0'(t)
	\]
	
	\noindent Подставляя $t=x$ убиваем остаток
	\[
	p'(x) = 
	\sum_{k=1}^{m} p(x_k) L_k'(x)
	\]
	
	\noindent Получили, что $\forall p \in \mathbb{P}_m$
	\begin{equation}
		\lambda^* p'(x) = 
		\sum_{k=1}^{m} p(x_k) ( \lambda^* L_k'(x) ) 
		\label{eq:re1}
	\end{equation}
	
	\noindent Но из условия \eqref{eq:cond1} 
	\[
	A(p) = 0, \quad \forall p \in \mathbb{P}_m
	\]
	
	\noindent То есть $\forall p \in \mathbb{P}_m$
	\begin{equation}
		\lambda^* p'(x) = 
		\sum_{k=1}^{m} p(x_k) ( a_k ) 
		\label{eq:re2}
	\end{equation}
	
	\noindent Из \eqref{eq:re1} и \eqref{eq:re2} получаем $\forall p \in \mathbb{P}_m$
	\[
	\sum_{k=1}^{m} p(x_k) ( a_k ) = \sum_{k=1}^{m} p(x_k) ( \lambda^* L_k'(x) ) 
	\] 
	
	\noindent Выбирая в качестве $p$ штуки которые равны $1$ и $0$ где надо, делаем вывод
	\[
	a_k = \lambda^* L_k'(x), \ \forall k \in 1 \ldots m 
	\]
	\[
	\omega_k (-1)^{k+1} = \lambda^* L_k'(x), \ \forall k \in 1 \ldots m 
	\]
	\[
	\omega_k  = (-1)^{k+1} \lambda^* L_k'(x), \ \forall k \in 1 \ldots m 
	\]
	
	
	\noindent Юзая условие на веса $\sum_{k=1}^{m} \omega_k = 1$ ищем лямбду:
	\[
	1 = \sum_{k=1}^{m} \omega_k = \sum_{k=1}^{m} (-1)^{k+1} \lambda^* L_k'(x) = \lambda^* \sum_{k=1}^{m} (-1)^{k+1}  L_k'(x)
	\]
	
	\noindent получаем
	\[
	\lambda^* = \left( \sum_{k=1}^{m} (-1)^{k+1}  L_k'(x) \right) ^ {-1}
	\]
	
	\noindent финальное выражение для весов
	\[
	\omega_k  = 
	\frac
	{
		(-1)^{k+1} L_k'(x)
	}
	{
		\sum_{i=1}^{m} (-1)^{i+1}  L_i'(x)
	}, \ \forall k \in 1 \ldots m 
	\]
	
	\[
	L_k(x) = \underset{k \neq j}{\prod_{j=1}^{m}} \frac{x - x_j}{x_k - x_j}, \ k \in 1 \ldots m
	\]

	\subsection{А теперь ищем узлы}
	
	\subsection*{Если растяжка вправо}
	
	\noindent В таком сетапе считаем
	\[
	-1 = x_1 < x_2 < \ldots < x_m < 1 < z
	\]
	
	\noindent где $x_k = a t_{k-1} + b, k \in 1, \ldots m$ \\
	и $z=a t_m + b$ (её игнорим)\\
	для неких $a, b \in \mathbb{R}$, где $a > 0$ (мы их не знаем, но они явно и не нужны, главное что какое-то линейное преобразование) \\
	а $t_k, \  k \in 0, \ldots, m$ есть точки альтернанса полинома Чебышёва 1-го рода 
	\[
	T_{m}(x) = \cos( m \arccos(x))
	\]
	\noindent Они имеют вид 
	\[
	t_k = \cos \left( \frac{ (m-k) \pi}{m} \right), \ k \in 0 \ldots m
	\]
	
	\noindent Уравнения описывающие растяжение (отношение длин интервалов между точками альтернанса сохраняется)
	\begin{align*}
		\frac{x_3 - x_2}{x_2 - x_1} &= \frac{t_3 - t_2}{t_2 - t_1}\\ 
		\frac{x_4 - x_3}{x_2 - x_1} &= \frac{t_4 - t_3}{t_2 - t_1}\\
		&\cdots\\
		\frac{x_m - x_{m-1}}{x_2 - x_1} &= \frac{t_m - t_{m-1}}{t_2 - t_1}\\
	\end{align*}
	\noindent уже известно, что $x_1 = -1$, значит из первого уравнения $x_3$ линейно выражается через $x_2$.\\
	из второго $x_4$ линейно выражается через $x_2$ и $x_3$ то есть линейно через $x_2$\\
	в конце концов все $x_k, \ k \in 1 \ldots m $ линейно выражаются через $x_2$\\
	Для окончательного нахождения плана достаточно найти $x_2$ 
	\textbf{ИЛИ} найти $a$ и $b$.\\
	
	
	\noindent Заметим, что узлы удовлетворяют
	\[
	p_0'(x) = 
	\frac{d}{dt} \left[\prod_{k=1}^{m} (t - x_k) \right]_{t=x} = 
	\frac{d}{dt} \left[\prod_{k=1}^{m} (t - (at_{k-1} + b) ) \right]_{t=x} = 
	0
	\]
	
	\noindent Введём новую переменную $s = \frac{t-b}{a}$
	\[
	\frac{d}{dt} \left[\prod_{k=1}^{m} (t - (at_{k-1} + b) ) \right] =
	\frac{1}{a}\frac{d}{dt} \left[\prod_{k=1}^{m} \left( \frac{t-b}{a} - t_{k-1} \right) \right] 
	\]
	\[
	\frac{d}{dt} \left[\prod_{k=1}^{m} \left( \frac{t-b}{a} - t_{k-1} \right) \right] = 0 
	\quad \Longleftrightarrow \quad
	\frac{d}{ds} \left[\prod_{k=1}^{m} \left( s - t_{k-1} \right) \right] = 0
	\]
	
	\noindent В $x_k = a t_{k-1} + b, k \in 1, \ldots m$ подставляем $k=1$ и получаем
	\[

	\frac
	{
		(-1)^{k+1} L_k'(x)
	}
	{
		\sum_{i=1}^{m} (-1)^{i+1}  L_i'(x)
	}, \ \forall k \in 1 \ldots m 
	\]
	
	\[
	L_k(x) = \underset{k \neq j}{\prod_{j=1}^{m}} \frac{x - x_j}{x_k - x_j}, \ k \in 1 \ldots m
	\]
	
	\subsection{Ограничение на $x$ от весов}
	
	Мы требуем $\forall k \in 1 \ldots m \ \omega_k > 0 $
	
	\noindent То есть нужно решить систему неравенств относительно $x$:
	\[
	\omega_k  = 
	(-1)^{k+1} \lambda^* L_k'(x)> 0 , \ \forall k \in 1 \ldots m 
	\]
	
	\subsection{Узлы}
	
	Теперь узлы должны быть расположены следующим образом:
	\[
	-1 = x_1 < x_2 < \ldots < x_{m-1} < x_m = 1
	\]
	
	
	\section{Решение системы}
	
	\subsection{Исходная система}
	
	Фиксируем $x\in\mathbb{R}$ и рассматриваем
	
	$$
	f(x)=
	\begin{pmatrix}
		1\\
		x\\
		x^2\\
		x^3\\
		x^4
	\end{pmatrix},
	\qquad
	c=f'(x)=
	\begin{pmatrix}
		0\\
		1\\
		2x\\
		3x^2\\
		4x^3
	\end{pmatrix}.
	$$
	
	Пусть
	
	$$
	\xi=
	\begin{pmatrix}
		x_1&x_2&x_3&x_4\\
		\omega_1&\omega_2&\omega_3&\omega_4
	\end{pmatrix}.
	$$
	
	При $m=n=4$ имеем
	
	$$
	\lambda^*
	\begin{pmatrix}
		0\\
		1\\
		2x\\
		3x^2\\
		4x^3
	\end{pmatrix}
	=
	\sum_{k=1}^4
	(-1)^{k+1}\omega_k
	\begin{pmatrix}
		1\\
		x_k\\
		x_k^2\\
		x_k^3\\
		x_k^4
	\end{pmatrix}.
	$$
	
	Фиксируем крайние узлы
	
	$$
	x_1=-1,
	\qquad
	x_4=1.
	$$
	
	Для удобства обозначим
	
	$$
	x_2=u,
	\qquad
	x_3=v.
	$$
	
	Таким образом,
	
	$$
	x_1=-1,\qquad
	x_2=u,\qquad
	x_3=v,\qquad
	x_4=1.
	$$
	
	\subsection{Система по компонентам}
	
	После покомпонентного раскрытия получаем
	
	$$
	\begin{cases}
		\omega_1-\omega_2+\omega_3-\omega_4=0,\\[2mm]
		-\omega_1-\omega_2u+\omega_3v-\omega_4=\lambda^*,\\[2mm]
		\omega_1-\omega_2u^2+\omega_3v^2-\omega_4=2\lambda^*x,\\[2mm]
		-\omega_1-\omega_2u^3+\omega_3v^3-\omega_4=3\lambda^*x^2,\\[2mm]
		\omega_1-\omega_2u^4+\omega_3v^4-\omega_4=4\lambda^*x^3.
	\end{cases}
	$$
	
	Дополнительно выполняется условие нормировки
	
	$$
	\omega_1+\omega_2+\omega_3+\omega_4=1.
	$$
	
	\subsection{Выражение $\omega_3$ и $\omega_4$}
	
	Из первого уравнения системы имеем
	
	$$
	\omega_3-\omega_4=\omega_2-\omega_1.
	$$
	
	Из условия нормировки
	
	$$
	\omega_3+\omega_4=1-\omega_1-\omega_2.
	$$
	
	Складывая и вычитая эти равенства, получаем
	
	$$
	\boxed{
		\omega_3=\frac12-\omega_1
	}
	$$
	
	и
	
	$$
	\boxed{
		\omega_4=\frac12-\omega_2.
	}
	$$
	
	Следовательно,
	
	$$
	\omega_1=\omega_1,\qquad
	\omega_2=\omega_2,\qquad
	\omega_3=\frac12-\omega_1,\qquad
	\omega_4=\frac12-\omega_2.
	$$
	
	\subsection{Исключение $\lambda^*$}
	
	Введём обозначение
	
	$$
	M_r=
	(-1)^r\omega_1
	-\omega_2u^r
	+\left(\frac12-\omega_1\right)v^r
	-\left(\frac12-\omega_2\right).
	$$
	
	Тогда четыре ненулевых уравнения исходной системы принимают вид
	
	$$
	M_1=\lambda^*,
	$$
	
	$$
	M_2=2\lambda^*x,
	$$
	
	$$
	M_3=3\lambda^*x^2,
	$$
	
	$$
	M_4=4\lambda^*x^3.
	$$
	
	Из первого равенства
	
	$$
	\lambda^*=M_1.
	$$
	
	Поэтому вместо остальных трёх уравнений можно рассматривать
	
	$$
	\boxed{
		F_1=M_2-2xM_1=0,
	}
	$$
	
	$$
	\boxed{
		F_2=M_3-3x^2M_1=0,
	}
	$$
	
	$$
	\boxed{
		F_3=M_4-4x^3M_1=0.
	}
	$$
	
	После раскрытия скобок получаем
	
	$$
	\boxed{
		F_1=
		\omega_1(1+2x)
		+\omega_2(2u-2x)
		+\frac{v^2-v}{2}
		-1+x
		=0,
	}
	$$
	
	$$
	\boxed{
		F_2=
		\omega_1(3x^2-1)
		+\omega_2(3x^2u-u^3)
		+\frac{v^3-3x^2v}{2}
		-1+3x^2
		=0,
	}
	$$
	
	$$
	\boxed{
		F_3=
		\omega_1(1+4x^3)
		+\omega_2(4x^3u-u^4)
		+\frac{v^4-4x^3v}{2}
		-1+4x^3
		=0.
	}
	$$
	
	\subsection{Базисные многочлены Лагранжа}
	
	Для узлов
	
	$$
	-1,\qquad u,\qquad v,\qquad 1
	$$
	
	базисные многочлены Лагранжа имеют вид
	
	$$
	L_k(t)
	=
	\prod_{\substack{j=1\\j\ne k}}^4
	\frac{t-x_j}{x_k-x_j}.
	$$
	
	Их производные в точке $x$ имеют вид
	
	$$
	\boxed{
		L_1'(x)=
		-\frac{
			uv-2ux+u-2vx+v+3x^2-2x
		}{
			2(u+1)(v+1)
		}.
	}
	$$
	
	Для второго узла
	
	$$
	\boxed{
		L_2'(x)=
		\frac{
			-2vx+v+3x^2-2x-1
		}{
			(u-1)(u+1)(u-v)
		}.
	}
	$$
	
	Для третьего узла
	
	$$
	\boxed{
		L_3'(x)=
		\frac{
			2ux-u-3x^2+2x+1
		}{
			(u-v)(v-1)(v+1)
		}.
	}
	$$
	
	Для четвёртого узла
	
	$$
	\boxed{
		L_4'(x)=
		\frac{
			uv-2ux-u-2vx-v+3x^2+2x
		}{
			2(u-1)(v-1)
		}.
	}
	$$
	
	\subsection{Выражение весов через $u$ и $v$}
	
	Используем формулу
	
	$$
	\omega_k=
	\frac{
		(-1)^{k+1}L_k'(x)
	}{
		\displaystyle
		\sum_{i=1}^4(-1)^{i+1}L_i'(x)
	}.
	$$
	
	Введём обозначение
	
	$$
	\boxed{
		D=
		L_1'(x)-L_2'(x)+L_3'(x)-L_4'(x).
	}
	$$
	
	Тогда
	
	$$
	\boxed{
		\omega_1=\frac{L_1'(x)}{D},
		\qquad
		\omega_2=-\frac{L_2'(x)}{D},
	}
	$$
	
	$$
	\boxed{
		\omega_3=\frac{L_3'(x)}{D},
		\qquad
		\omega_4=-\frac{L_4'(x)}{D}.
	}
	$$
	
	То есть
	
	$$
	\boxed{
		\omega_1=
		-\frac{
			uv-2ux+u-2vx+v+3x^2-2x
		}{
			2(u+1)(v+1)D
		}.
	}
	$$
	
	$$
	\boxed{
		\omega_2=
		-\frac{
			-2vx+v+3x^2-2x-1
		}{
			(u-1)(u+1)(u-v)D
		}.
	}
	$$
	
	$$
	\boxed{
		\omega_3=
		\frac{
			2ux-u-3x^2+2x+1
		}{
			(u-v)(v-1)(v+1)D
		}.
	}
	$$
	
	$$
	\boxed{
		\omega_4=
		-\frac{
			uv-2ux-u-2vx-v+3x^2+2x
		}{
			2(u-1)(v-1)D
		}.
	}
	$$
	
	\subsection{Подстановка весов в систему}
	
	Подставляя найденные выражения для весов в
	
	$$
	F_1=0,\qquad
	F_2=0,\qquad
	F_3=0,
	$$
	
	после сокращения общих множителей получаем
	
	$$
	\boxed{
		F_1\equiv0,
	}
	$$
	
	$$
	\boxed{
		F_2\equiv0.
	}
	$$
	
	Из третьего уравнения остаётся нетривиальное условие
	
	$$
	\boxed{
		2uvx-3uv
		-3ux^2+2ux+2u
		-3vx^2+2vx+2v
		+4x^3-3x^2-2x
		=0.
	}
	$$
	
	Таким образом, это единственное дополнительное уравнение на $u$ и $v$, которое остаётся после использования формулы оптимальных весов.
	
	\subsection{Условие $x_3-x_2=\alpha$}
	
	Добавим дополнительное условие
	
	$$
	x_3-x_2=\alpha.
	$$
	
	Поскольку
	
	$$
	x_2=u,\qquad x_3=v,
	$$
	
	получаем
	
	$$
	v-u=\alpha.
	$$
	
	Следовательно,
	
	$$
	\boxed{
		v=u+\alpha.
	}
	$$
	
	Таким образом,
	
	$$
	\boxed{
		x_2=u,\qquad
		x_3=u+\alpha.
	}
	$$
	
	\subsection{Подстановка $v=u+\alpha$}
	
	Подставляем
	
	$$
	v=u+\alpha
	$$
	
	в уравнение
	
	$$
	2uvx-3uv
	-3ux^2+2ux+2u
	-3vx^2+2vx+2v
	+4x^3-3x^2-2x
	=0.
	$$
	
	Получаем
	
	$$
	\begin{aligned}
		&2u(u+\alpha)x
		-3u(u+\alpha)
		-3ux^2
		+2ux
		+2u
		\\
		&\qquad
		-3(u+\alpha)x^2
		+2(u+\alpha)x
		+2(u+\alpha)
		+4x^3-3x^2-2x
		=0.
	\end{aligned}
	$$
	
	Собирая слагаемые по степеням $u$, получаем
	
	$$
	\boxed{
		(2x-3)u^2
		+
		\left(
		2\alpha x-3\alpha
		-6x^2+4x+4
		\right)u
		+
		\left(
		-3\alpha x^2
		+2\alpha x
		+2\alpha
		+4x^3-3x^2-2x
		\right)
		=0.
	}
	$$
	
	\subsection{Введение $A$, $B$, $C$}
	
	Для компактности обозначим
	
	$$
	\boxed{
		A=2x-3,
	}
	$$
	
	$$
	\boxed{
		B=
		2\alpha x-3\alpha
		-6x^2+4x+4,
	}
	$$
	
	$$
	\boxed{
		C=
		-3\alpha x^2
		+2\alpha x
		+2\alpha
		+4x^3-3x^2-2x.
	}
	$$
	
	Тогда уравнение для $u=x_2$ принимает стандартный вид
	
	$$
	\boxed{
		Au^2+Bu+C=0.
	}
	$$
	
	При условии
	
	$$
	2x-3\ne0
	$$
	
	получаем
	
	$$
	\boxed{
		x_2=u=
		\frac{-B\pm\sqrt{B^2-4AC}}{2A}.
	}
	$$
	
	Непосредственно через $x$ и $\alpha$:
	
	$$
	\boxed{
		x_2=
		\frac{
			-\left(
			2\alpha x-3\alpha-6x^2+4x+4
			\right)
			\pm
			\sqrt{
				\left(
				2\alpha x-3\alpha-6x^2+4x+4
				\right)^2
				-
				4(2x-3)
				\left(
				-3\alpha x^2+2\alpha x+2\alpha
				+4x^3-3x^2-2x
				\right)
			}
		}{
			2(2x-3)
		}.
	}
	$$
	
	После этого
	
	$$
	\boxed{
		x_3=x_2+\alpha.
	}
	$$
	
	Или, используя обозначения $A,B,C$,
	
	$$
	\boxed{
		x_3=
		\frac{-B\pm\sqrt{B^2-4AC}}{2A}
		+\alpha.
	}
	$$
	
	Знак $\pm$ должен быть выбран согласованно для $x_2$ и $x_3$.
	
	\subsection{Восстановление весов}
	
	После нахождения $x_2$ и $x_3$ подставляем их в
	
	$$
	D=
	L_1'(x)-L_2'(x)+L_3'(x)-L_4'(x).
	$$
	
	Тогда веса определяются формулами
	
	$$
	\boxed{
		\omega_1=\frac{L_1'(x)}{D},
		\qquad
		\omega_2=-\frac{L_2'(x)}{D},
	}
	$$
	
	$$
	\boxed{
		\omega_3=\frac{L_3'(x)}{D},
		\qquad
		\omega_4=-\frac{L_4'(x)}{D}.
	}
	$$
	
	При этом выполняются соотношения
	
	$$
	\boxed{
		\omega_3=\frac12-\omega_1,
		\qquad
		\omega_4=\frac12-\omega_2.
	}
	$$
	
	\subsection{Восстановление $\lambda^*$}
	
	Из первого ненулевого уравнения исходной системы
	
	$$
	-\omega_1-\omega_2x_2+\omega_3x_3-\omega_4
	=
	\lambda^*
	$$
	
	получаем
	
	$$
	\boxed{
		\lambda^*
		=
		-\omega_1-\omega_2x_2+\omega_3x_3-\omega_4.
	}
	$$
	
	С учётом
	
	$$
	x_2=u,\qquad
	x_3=u+\alpha
	$$
	
	имеем
	
	$$
	\boxed{
		\lambda^*
		=
		-\omega_1-\omega_2u
		+\omega_3(u+\alpha)
		-\omega_4.
	}
	$$
	
	Используя
	
	$$
	\omega_3=\frac12-\omega_1,
	\qquad
	\omega_4=\frac12-\omega_2,
	$$
	
	получаем
	
	$$
	\boxed{
		\lambda^*
		=
		-\omega_1-\omega_2u
		+\left(\frac12-\omega_1\right)(u+\alpha)
		-\left(\frac12-\omega_2\right).
	}
	$$
	
	\subsection{Итоговая схема решения}
	
	Таким образом,
	
	$$
	\boxed{
		x_1=-1,\qquad
		x_4=1,\qquad
		x_3-x_2=\alpha.
	}
	$$
	
	Полагая
	
	$$
	x_2=u,\qquad
	x_3=u+\alpha,
	$$
	
	получаем
	
	$$
	\boxed{
		x_2=u,\qquad
		x_3=u+\alpha.
	}
	$$
	
	Значение $u$ определяется из квадратного уравнения
	
	$$
	\boxed{
		(2x-3)u^2
		+
		(2\alpha x-3\alpha-6x^2+4x+4)u
		+
		(-3\alpha x^2+2\alpha x+2\alpha
		+4x^3-3x^2-2x)
		=0.
	}
	$$
	
	При $2x-3\ne0$
	
	$$
	\boxed{
		x_2=
		\frac{-B\pm\sqrt{B^2-4AC}}{2A},
		\qquad
		x_3=x_2+\alpha,
	}
	$$
	
	где
	
	$$
	\boxed{
		A=2x-3,
	}
	$$
	
	$$
	\boxed{
		B=2\alpha x-3\alpha-6x^2+4x+4,
	}
	$$
	
	$$
	\boxed{
		C=-3\alpha x^2+2\alpha x+2\alpha
		+4x^3-3x^2-2x.
	}
	$$
	
	После определения узлов вычисляются веса:
	
	$$
	\boxed{
		\omega_k=
		\frac{
			(-1)^{k+1}L_k'(x)
		}{
			L_1'(x)-L_2'(x)+L_3'(x)-L_4'(x)
		}.
	}
	$$
	
	Наконец,
	
	$$
	\boxed{
		\lambda^*
		=
		-\omega_1-\omega_2x_2+\omega_3x_3-\omega_4.
	}
	$$
	
	При выборе ветви $\pm$ необходимо учитывать допустимость узлов
	
	$$
	\boxed{
		-1<x_2<x_3<1
	}
	$$
	
	и ненулевость знаменателей, возникающих в формулах для многочленов Лагранжа.
